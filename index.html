/** * Google Apps Script Backend for Plickers Web App. * Attach this script to your Google Spreadsheet (Extensions > Apps Script). * Deploy as Web App: Execute as "Me", Who has access: "Anyone". */ function doGet(e) { var data = getData(); return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var postData = JSON.parse(e.postData.contents); var action = postData.action; if (action === 'setActiveQuestion') { setActiveQuestion(postData.questionId, postData.folderName); } else if (action === 'saveResponse') { saveResponse(postData.questionId, postData.studentId, postData.selectedAnswer, postData.folderName); } else if (action === 'addBulkQuestions') { addBulkQuestions(postData.folderName, postData.questionsList); } else if (action === 'deleteQuestion') { deleteQuestion(postData.questionId, postData.folderName); } else if (action === 'deleteFolder') { deleteFolder(postData.folderName); } else if (action === 'deleteResponse') { deleteResponse(postData.questionId, postData.studentId, postData.folderName); } else if (action === 'resetQuestionResponses') { resetQuestionResponses(postData.questionId, postData.folderName); } else if (action === 'clearHistory') { clearHistory(); } else if (action === 'saveStudentNames') { saveStudentNames(postData.studentList); } var data = getData(); return ContentService.createTextOutput(JSON.stringify({ status: "success", data: data })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function getSheet(name) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); } return sheet; } function getData() { var qSheet = getSheet('Questions'); var rSheet = getSheet('Responses'); var sSheet = getSheet('State'); var rosSheet = getSheet('Roster'); var qData = qSheet.getDataRange().getValues(); var questions = []; var foldersSet = {}; for (var i = 1; i < qData.length; i++) { if (qData[i][0]) { var folder = qData[i][7] || "General"; foldersSet[folder] = true; questions.push({ id: String(qData[i][0]), text: qData[i][1], a: qData[i][2], b: qData[i][3], c: qData[i][4], d: qData[i][5], correct: String(qData[i][6]).toUpperCase(), folder: folder, explanation: qData[i][8] || "" }); } } var activeQ = sSheet.getRange("A1").getValue() || ""; var activeFolder = sSheet.getRange("B1").getValue() || ""; var respData = rSheet.getDataRange().getValues(); var logs = []; var allLogs = []; for (var j = 1; j < respData.length; j++) { if (respData[j][0]) { var qId = String(respData[j][0]); var sId = String(respData[j][1]); var ans = String(respData[j][2]).toUpperCase(); var respFolder = respData[j][4] ? String(respData[j][4]) : ""; var qObj = questions.find(function(q) { return q.id === qId && (respFolder ? q.folder === respFolder : (!activeFolder || q.folder === activeFolder)); }); if (!qObj) { qObj = questions.find(function(q) { return q.id === qId; }); } var isCorrect = (qObj && qObj.correct === ans) ? "Correct" : "Incorrect"; var entry = { questionId: qId, studentId: sId, selectedAnswer: ans, isCorrect: isCorrect, folder: respFolder }; allLogs.push(entry); if (qId === String(activeQ) && (!activeFolder || !respFolder || respFolder === String(activeFolder))) { logs.push(entry); } } } var rosData = rosSheet.getDataRange().getValues(); var studentList = []; for (var k = 1; k < rosData.length; k++) { if (rosData[k][0]) { studentList.push({ id: String(rosData[k][0]), name: rosData[k][1] }); } } return { questions: questions, folders: Object.keys(foldersSet), activeQuestion: String(activeQ), activeFolder: String(activeFolder), logs: logs, allLogs: allLogs, studentList: studentList }; } function setActiveQuestion(qId, folderName) { var sSheet = getSheet('State'); sSheet.getRange("A1").setValue(String(qId)); if (folderName) { sSheet.getRange("B1").setValue(String(folderName)); } } function saveResponse(qId, studentId, selectedAnswer, folderName) { var rSheet = getSheet('Responses'); var data = rSheet.getDataRange().getValues(); var foundIndex = -1; for (var i = 1; i < data.length; i++) { var matchQ = (String(data[i][0]) === String(qId)); var matchS = (String(data[i][1]) === String(studentId)); var matchF = (!folderName || !data[i][4] || String(data[i][4]) === String(folderName)); if (matchQ && matchS && matchF) { foundIndex = i + 1; break; } } if (foundIndex > 0) { rSheet.getRange(foundIndex, 3).setValue(selectedAnswer); if (folderName) rSheet.getRange(foundIndex, 5).setValue(folderName); } else { rSheet.appendRow([qId, studentId, selectedAnswer, new Date(), folderName || ""]); } } function addBulkQuestions(folderName, questionsList) { var qSheet = getSheet('Questions'); if (qSheet.getLastRow() === 0) { qSheet.appendRow(["ID", "Text", "A", "B", "C", "D", "Correct", "Folder", "Explanation"]); } questionsList.forEach(function(q) { qSheet.appendRow([q.id, q.text, q.a, q.b, q.c, q.d, q.correct, folderName, q.explanation]); }); } function deleteQuestion(qId, folderName) { var qSheet = getSheet('Questions'); var data = qSheet.getDataRange().getValues(); for (var i = data.length - 1; i >= 1; i--) { var rowId = String(data[i][0]).trim(); var rowFolder = String(data[i][7] || "General").trim(); var matchId = (rowId === String(qId).trim()); var matchFolder = (!folderName || rowFolder === String(folderName).trim()); if (matchId && matchFolder) { qSheet.deleteRow(i + 1); break; } } } function deleteFolder(folderName) { var qSheet = getSheet('Questions'); var data = qSheet.getDataRange().getValues(); for (var i = data.length - 1; i >= 1; i--) { if (String(data[i][7] || "General") === String(folderName)) { qSheet.deleteRow(i + 1); } } } function deleteResponse(qId, studentId, folderName) { var rSheet = getSheet('Responses'); var data = rSheet.getDataRange().getValues(); for (var i = data.length - 1; i >= 1; i--) { var matchQ = (String(data[i][0]) === String(qId)); var matchS = (String(data[i][1]) === String(studentId)); var matchF = (!folderName || !data[i][4] || String(data[i][4]) === String(folderName)); if (matchQ && matchS && matchF) { rSheet.deleteRow(i + 1); } } } function resetQuestionResponses(qId, folderName) { var rSheet = getSheet('Responses'); var data = rSheet.getDataRange().getValues(); for (var i = data.length - 1; i >= 1; i--) { var matchQ = (String(data[i][0]) === String(qId)); var matchF = (!folderName || !data[i][4] || String(data[i][4]) === String(folderName)); if (matchQ && matchF) { rSheet.deleteRow(i + 1); } } } function clearHistory() { var rSheet = getSheet('Responses'); rSheet.clear(); rSheet.appendRow(["QuestionID", "StudentID", "SelectedAnswer", "Timestamp", "Folder"]); } function saveStudentNames(studentList) { var rosSheet = getSheet('Roster'); rosSheet.clear(); rosSheet.appendRow(["CardID", "StudentName"]); studentList.forEach(function(s) { rosSheet.appendRow([s.id, s.name]); }); }
